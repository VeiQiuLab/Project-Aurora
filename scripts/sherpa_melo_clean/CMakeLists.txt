cmake_minimum_required(VERSION 3.24)
project(aurora_melo_feasibility LANGUAGES CXX)
# Experimental only: no upstream root build, downloads, or production wiring.
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This audited recipe requires MSVC Windows x64")
endif()
foreach(required SHERPA_SOURCE KALDIFST_SOURCE OPENFST_SOURCE ORT_HEADERS REFERENCE_RUNTIME)
  if(NOT IS_DIRECTORY "${${required}}")
    message(FATAL_ERROR "Supply -D${required}=external source/runtime directory")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_compile_options(/utf-8 /bigobj /Gy)
add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
foreach(feature BIN SCRIPT COMPACT COMPRESS CONST FAR LINEAR LOOKAHEAD MPDT NGRAM PDT SPECIAL)
  set(HAVE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(OPENFST_USE_ABSL OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPENFST_SOURCE}" "${CMAKE_BINARY_DIR}/openfst" EXCLUDE_FROM_ALL)
set(kaldi_sources kaldi-fst-io kaldi-holder kaldi-io kaldi-math kaldi-semaphore kaldi-table parse-options text-normalizer text-utils)
foreach(source IN LISTS kaldi_sources)
  list(APPEND normalizer_sources "${KALDIFST_SOURCE}/kaldifst/csrc/${source}.cc")
endforeach()
add_library(normalizer STATIC ${normalizer_sources})
target_include_directories(normalizer PUBLIC "${KALDIFST_SOURCE}")
target_link_libraries(normalizer PUBLIC fst)
# Match upstream Windows optimizer workaround only for this complex FST TU.
set_source_files_properties("${KALDIFST_SOURCE}/kaldifst/csrc/text-normalizer.cc" PROPERTIES COMPILE_OPTIONS "/Od;/Ob0")
set(sherpa_sources offline-tts-vits-model melo-tts-lexicon phrase-matcher symbol-table text-utils file-utils onnx-utils offline-tts-frontend lexicon base64-decode bbpe)
foreach(source IN LISTS sherpa_sources)
  list(APPEND inference_sources "${SHERPA_SOURCE}/sherpa-onnx/csrc/${source}.cc")
endforeach()
# Compile the original silence utility verbatim, without OfflineTts::Create's
# all-model dispatch. Generated third-party source stays outside the repository.
file(READ "${SHERPA_SOURCE}/sherpa-onnx/csrc/offline-tts.cc" audio_source)
string(FIND "${audio_source}" "std::string GenerationConfig::GetExtraString" cutoff)
if(cutoff LESS 1)
  message(FATAL_ERROR "Pinned source layout changed")
endif()
string(SUBSTRING "${audio_source}" 0 ${cutoff} audio_source)
string(REPLACE "#include \"sherpa-onnx/csrc/offline-tts-impl.h\"" "" audio_source "${audio_source}")
file(WRITE "${CMAKE_BINARY_DIR}/silence.cc" "${audio_source}\n}  // namespace sherpa_onnx\n")
add_library(clean_melo SHARED clean_c_api.cpp cpu_session.cpp ${inference_sources} "${CMAKE_BINARY_DIR}/silence.cc")
target_compile_definitions(clean_melo PRIVATE SHERPA_ONNX_BUILD_SHARED_LIBS SHERPA_ONNX_BUILD_MAIN_LIB)
target_include_directories(clean_melo PRIVATE "${SHERPA_SOURCE}" "${ORT_HEADERS}")
target_link_libraries(clean_melo PRIVATE normalizer "${REFERENCE_RUNTIME}/lib/onnxruntime.lib")
target_link_options(clean_melo PRIVATE /OPT:REF /OPT:ICF)
set_target_properties(clean_melo PROPERTIES OUTPUT_NAME sherpa-onnx-c-api
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/clean-runtime/lib"
  ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/clean-runtime/lib"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/clean-runtime/lib"
  ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/clean-runtime/lib")
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(benchmark_sherpa_melo "${repo}/scripts/benchmark_sherpa_melo.cpp")
target_compile_definitions(benchmark_sherpa_melo PRIVATE SHERPA_ONNX_BUILD_SHARED_LIBS)
target_include_directories(benchmark_sherpa_melo PRIVATE "${REFERENCE_RUNTIME}/include")
target_link_libraries(benchmark_sherpa_melo PRIVATE clean_melo psapi)
set_target_properties(benchmark_sherpa_melo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/clean-bin")
foreach(dll onnxruntime.dll onnxruntime_providers_shared.dll)
  add_custom_command(TARGET clean_melo POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${REFERENCE_RUNTIME}/lib/${dll}" "$<TARGET_FILE_DIR:clean_melo>/${dll}")
  add_custom_command(TARGET benchmark_sherpa_melo POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${REFERENCE_RUNTIME}/lib/${dll}" "$<TARGET_FILE_DIR:benchmark_sherpa_melo>/${dll}")
endforeach()
add_custom_command(TARGET benchmark_sherpa_melo POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:clean_melo>" "$<TARGET_FILE_DIR:benchmark_sherpa_melo>/sherpa-onnx-c-api.dll")
# Deliberately separate oracle targets; NEVER a dependency of the clean host.
file(READ "${repo}/scripts/benchmark_sherpa_melo.cpp" oracle_source)
string(REPLACE "config.model.num_threads = threads;" "config.model.debug = 1; config.model.num_threads = threads;" oracle_source "${oracle_source}")
file(WRITE "${CMAKE_BINARY_DIR}/oracle-main.cpp" "${oracle_source}")
foreach(kind reference clean)
  add_executable(${kind}_oracle EXCLUDE_FROM_ALL "${CMAKE_BINARY_DIR}/oracle-main.cpp" oracle_trace.cpp)
  target_compile_definitions(${kind}_oracle PRIVATE SHERPA_ONNX_BUILD_SHARED_LIBS)
  target_include_directories(${kind}_oracle PRIVATE "${REFERENCE_RUNTIME}/include" "${ORT_HEADERS}")
  target_link_libraries(${kind}_oracle PRIVATE "${REFERENCE_RUNTIME}/lib/onnxruntime.lib" psapi)
  set_target_properties(${kind}_oracle PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${kind}-oracle")
  if(kind STREQUAL "reference")
    target_link_libraries(${kind}_oracle PRIVATE "${REFERENCE_RUNTIME}/lib/sherpa-onnx-c-api.lib")
    set(api_dll "${REFERENCE_RUNTIME}/lib/sherpa-onnx-c-api.dll")
  else()
    target_link_libraries(${kind}_oracle PRIVATE clean_melo)
    set(api_dll "$<TARGET_FILE:clean_melo>")
  endif()
  add_custom_command(TARGET ${kind}_oracle POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${api_dll}" "$<TARGET_FILE_DIR:${kind}_oracle>/sherpa-onnx-c-api.dll")
  foreach(dll onnxruntime.dll onnxruntime_providers_shared.dll)
    add_custom_command(TARGET ${kind}_oracle POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${REFERENCE_RUNTIME}/lib/${dll}" "$<TARGET_FILE_DIR:${kind}_oracle>/${dll}")
  endforeach()
endforeach()
foreach(forbidden espeak-ng espeak-ng-for-piper piper-phonemize sherpa-onnx-core kaldi-decoder)
  if(TARGET ${forbidden})
    message(FATAL_ERROR "Forbidden broad/GPL target: ${forbidden}")
  endif()
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/clean-source-manifest.txt" "${inference_sources};${normalizer_sources}\nGPL targets absent; upstream root not configured; no FetchContent.\n")
