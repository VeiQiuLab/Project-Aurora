cmake_minimum_required(VERSION 3.16)
project(AuroraLive2DAdapter LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The minimal native surface requires Windows/MSVC")
endif()
set(CUBISM_SDK_ROOT "" CACHE PATH "Read-only Cubism Native 5-r.5 SDK")
file(READ "${CUBISM_SDK_ROOT}/cubism-info.yml" SDK_INFO)
if(NOT SDK_INFO MATCHES "version: 5-r[.]5")
  message(FATAL_ERROR "Expected Cubism Native 5-r.5")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(FRAMEWORK_SOURCE D3D11)
add_subdirectory("${CUBISM_SDK_ROOT}/Framework" "${CMAKE_BINARY_DIR}/framework")
target_compile_definitions(Framework PUBLIC NOMINMAX)
target_include_directories(Framework PUBLIC "${CUBISM_SDK_ROOT}/Core/include")
target_link_libraries(Framework PUBLIC
  "${CUBISM_SDK_ROOT}/Core/lib/windows/x86_64/143/Live2DCubismCore_MD$<$<CONFIG:Debug>:d>.lib"
  d3d11 d3dcompiler)
add_library(aurora_live2d SHARED adapter.cpp)
target_compile_definitions(aurora_live2d PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(aurora_live2d PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(aurora_live2d PRIVATE Framework dxgi dcomp windowscodecs ole32 user32)
