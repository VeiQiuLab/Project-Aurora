cmake_minimum_required(VERSION 3.24)
project(aurora_local_voice LANGUAGES CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Requires MSVC Windows x64")
endif()
# Consume only the independently audited minimal Melo ABI, never the broad
# official sherpa TTS DLL (which links the excluded phonemizers).
foreach(input CLEAN_RUNTIME SHERPA_HEADERS)
  if(NOT IS_DIRECTORY "${${input}}")
    message(FATAL_ERROR "Supply external ${input}")
  endif()
endforeach()
file(SHA256 "${CLEAN_RUNTIME}/sherpa-onnx-c-api.dll" runtime_hash)
if(NOT runtime_hash STREQUAL "bb146780c7b946755810f28a5fb7dcf6321ae0d6bf18f7b507461a79c3396a3b")
  message(FATAL_ERROR "Audited minimal runtime hash mismatch")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_executable(aurora-local-voice-host host.cpp)
target_compile_options(aurora-local-voice-host PRIVATE /utf-8 /W4)
target_compile_definitions(aurora-local-voice-host PRIVATE SHERPA_ONNX_BUILD_SHARED_LIBS)
target_include_directories(aurora-local-voice-host PRIVATE "${SHERPA_HEADERS}")
target_link_libraries(aurora-local-voice-host PRIVATE "${CLEAN_RUNTIME}/sherpa-onnx-c-api.lib" psapi)
foreach(dll sherpa-onnx-c-api.dll onnxruntime.dll onnxruntime_providers_shared.dll)
  add_custom_command(TARGET aurora-local-voice-host POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CLEAN_RUNTIME}/${dll}" "$<TARGET_FILE_DIR:aurora-local-voice-host>/${dll}")
endforeach()
